**Tech Stack:** Go 1.26.5 / gotd v0.140.0（`tg.MessageReplies`/`MessageReactions`/`MessagesGetReplies`）/ Vue 3.5 / Element Plus / lucide-vue-next / vite。

```go
func messageInfoFrom(m *tg.Message) *MessageInfo {
	media, hasMedia := tmedia.GetMedia(m)
	info := &MessageInfo{
		ID:       m.ID,
		Date:     m.Date,
		Text:     m.Message,
		Views:    m.Views,
		Forwards: m.Forwards,
		Out:      m.Out,
		HasMedia: hasMedia,
	}
	if gid, ok := m.GetGroupedID(); ok {
		info.GroupedID = gid
	}
	if lp := linkPreviewFromMessage(m); lp != nil {
		info.LinkPreview = lp
	}
	if hasMedia {
		info.Media = messageMediaFrom(m, media)
	}
	if re := repliesFrom(m); re != nil {
		info.Replies = re
	}
	if rc := reactionsFrom(m); len(rc) > 0 {
		info.Reactions = rc
	}
	return info
}

// messageMediaFrom builds the downloadable-media struct for a message that
// carries media. Shared by messageInfoFrom and commentInfoFrom (Task 2).
func messageMediaFrom(m *tg.Message, media tmedia.Media) *MessageMedia {
	mm := &MessageMedia{Name: media.Name, Size: media.Size, DC: media.DC}
	mm.Width, mm.Height = mediaDimensions(m)
	mm.Mime = mediaMime(m)
	mm.Thumb = extractThumb(m)
	if st := tmedia.GetStrippedThumb(m); st != nil {
		mm.Stripped = &StrippedThumb{DataURI: st.DataURI, W: st.W, H: st.H}
	}
	return mm
}

// repliesFrom extracts a message's comment-section info. nil when the message
// has no replies at all. ChannelID is conditional in gotd — read via helper.
func repliesFrom(m *tg.Message) *MessageRepliesInfo {
	re, ok := m.GetReplies()
	if !ok {
		return nil
	}
	info := &MessageRepliesInfo{Count: re.Replies, HasComments: re.Comments}
	if cid, ok := re.GetChannelID(); ok {
		info.ChannelID = cid
	}
	if info.Count == 0 && !info.HasComments {
		return nil // no real comment section
	}
	return info
}

// reactionsFrom extracts a message's emoji reactions. Premium custom-emoji
// reactions (ReactionCustomEmoji) are skipped — v1 doesn't render them.
func reactionsFrom(m *tg.Message) []MessageReaction {
	re, ok := m.GetReactions()
	if !ok {
		return nil
	}
	var out []MessageReaction
	for _, rc := range re.Results {
		if em, ok := rc.Reaction.(*tg.ReactionEmoji); ok {
			out = append(out, MessageReaction{Emoji: em.Emoticon, Count: rc.Count})
		}
	}
	return out
}
```

```go
package chat

import (
	"context"

	"github.com/go-faster/errors"
	"github.com/gotd/td/telegram"
	"github.com/gotd/td/telegram/peers"
	"github.com/gotd/td/tg"

	"github.com/iyear/tdl/core/storage"
	"github.com/iyear/tdl/core/tmedia"
	"github.com/iyear/tdl/core/util/tutil"
)

// CommentInfo is one reply in a channel post's comment section. Author info
// comes from the getReplies response's own Users/Chats entities — no extra RPC.
type CommentInfo struct {
	ID        int               `json:"id"`
	Date      int               `json:"date"`
	Text      string            `json:"text"`
	AuthorID  int64             `json:"authorId"`
	Name      string            `json:"name"`                // user visibleName or chat/channel title
	Username  string            `json:"username,omitempty"`  // empty for groups/channels without username
	HasMedia  bool              `json:"hasMedia"`
	Media     *MessageMedia     `json:"media,omitempty"`
	Reactions []MessageReaction `json:"reactions,omitempty"`
}

// FetchReplies pages a channel post's comment section (newest first).
// chatID is the post's original channel; msgID is the post. offsetID is the
// cursor: pass 0 for the latest page, otherwise the smallest comment id from
// the previous page (exclusive — the server maps the channel post to its
// discussion group automatically, same as tdl's export --thread path).
// limit<=0 or >100 falls back to 50. Returns up to limit comments and the
// next cursor (0 = no more).
func FetchReplies(ctx context.Context, c *telegram.Client, kvd storage.Storage,
	chatID string, msgID, offsetID, limit int) ([]*CommentInfo, int, error) {

	if limit <= 0 || limit > 100 {
		limit = 50
	}

	manager := peers.Options{Storage: storage.NewPeers(kvd)}.Build(c.API())
	peer, err := tutil.GetInputPeer(ctx, manager, chatID)
	if err != nil {
		return nil, 0, errors.Wrap(err, "resolve chat")
	}

	res, err := c.API().MessagesGetReplies(ctx, &tg.MessagesGetRepliesRequest{
		Peer:       peer.InputPeer(),
		MsgID:      msgID,
		OffsetID:   offsetID,
		OffsetDate: 0,
		AddOffset:  0,
		Limit:      limit,
	})
	if err != nil {
		return nil, 0, errors.Wrap(err, "get replies")
	}

	ch, ok := res.(*tg.MessagesChannelMessages)
	if !ok {
		return nil, 0, errors.New("unexpected replies response")
	}

	// 作者实体跟随响应自带，零额外 RPC
	users := make(map[int64]*tg.User, len(ch.Users))
	for _, u := range ch.Users {
		if user, ok := u.(*tg.User); ok {
			users[user.ID] = user
		}
	}
	chats := make(map[int64]tg.ChatClass, len(ch.Chats))
	for _, cc := range ch.Chats {
		chats[cc.GetID()] = cc
	}

	out := make([]*CommentInfo, 0, len(ch.Messages))
	next := 0
	for i, mc := range ch.Messages {
		m, ok := mc.(*tg.Message)
		if !ok {
			continue
		}
		out = append(out, commentInfoFrom(m, users, chats))
		if i == len(ch.Messages)-1 {
			next = m.ID // 新→旧，最后一条最旧，作下一页排他游标
		}
	}
	if len(out) < limit {
		next = 0 // 不足一页 = 到底
	}
	return out, next, nil
}

// commentInfoFrom converts a comment message into CommentInfo, resolving the
// author from the response's entity maps. users/chats may be nil/empty.
func commentInfoFrom(m *tg.Message, users map[int64]*tg.User, chats map[int64]tg.ChatClass) *CommentInfo {
	info := &CommentInfo{
		ID:   m.ID,
		Date: m.Date,
		Text: m.Message,
	}
	if from, ok := m.GetFromID(); ok {
		info.AuthorID = tutil.GetPeerID(from)
		info.Name, info.Username = authorFrom(from, users, chats)
	}
	if media, has := tmedia.GetMedia(m); has {
		info.HasMedia = true
		info.Media = messageMediaFrom(m, media)
	}
	if rc := reactionsFrom(m); len(rc) > 0 {
		info.Reactions = rc
	}
	return info
}

// authorFrom resolves a comment author's display name + username from the
// response entities. Users get visibleName(first,last); chats/channels get
// their title (+username). Unknown authors fall back to empty strings.
func authorFrom(from tg.PeerClass, users map[int64]*tg.User, chats map[int64]tg.ChatClass) (name, username string) {
	id := tutil.GetPeerID(from)
	switch from.(type) {
	case *tg.PeerUser:
		if u, ok := users[id]; ok {
			return visibleName(u.FirstName, u.LastName), u.Username
		}
	case *tg.PeerChat, *tg.PeerChannel:
		if cc, ok := chats[id]; ok {
			switch c := cc.(type) {
			case *tg.Chat:
				return c.Title, ""
			case *tg.Channel:
				return c.Title, c.Username
			}
		}
	}
	return "", ""
}
```

```go
// handleChatReplies pages a channel post's comment section (newest first,
// read-only). Query: offset_id (cursor), limit. 无缓存无单飞——锚点相关的
// 查询，高频缓存收益低、正确性风险高（消息被删/新回复），同 handleChatAround 理由。
func (s *Server) handleChatReplies(w http.ResponseWriter, r *http.Request) {
	a := s.activeAccount()
	if a == nil {
		writeErr(w, http.StatusBadRequest, errString("尚未登录账号"))
		return
	}
	chatID, msgID, err := parseMediaVars(r)
	if err != nil {
		writeErr(w, http.StatusBadRequest, err)
		return
	}
	offsetID, _ := strconv.Atoi(r.URL.Query().Get("offset_id"))
	limit, _ := strconv.Atoi(r.URL.Query().Get("limit"))

	comments, next, err := chat.FetchReplies(r.Context(), a.client, a.kvd, chatID, msgID, offsetID, limit)
	if err != nil {
		s.log().Warn("fetch replies failed", zap.String("chat", chatID), zap.Int("msg", msgID), zap.Error(err))
		writeErr(w, http.StatusServiceUnavailable, errString("连接重连中，请稍候"))
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"comments":   comments,
		"nextCursor": next,
	})
}
```

```vue
async function loadMore() {
  if (!hasMore.value || loadingMore.value) return
  loadingMore.value = true
  try {
    const res = await loadPage(nextCursor.value)

    loadingMore.value = false
  }
}

  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`
}

    loading.value = false
  }
})
```
